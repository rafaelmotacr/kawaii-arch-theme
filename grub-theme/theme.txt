# Boot menu
# Somente icone + nome de cada sistema. Icones mantidos em 75x75.
# A largura e fixa em pixels (as fontes do GRUB tambem sao em pixels),
# entao a selecao fica do tamanho do maior texto em qualquer resolucao:
#   3 (padding) + 75 (icone) + 20 (espaco) + ~390 (texto) + 3 (padding)
# 390 px comporta "Advanced options for Arch Linux" em Public Sans Bold 21.
#
# Contorno "liquid glass" (boot_menu_box_*.png, mesmo contorno da caixa do
# terminal, sem preenchimento; gerado por tools/make_glass.py). O GRUB
# desenha essa moldura DENTRO dos limites do menu e reserva 16 px (tamanho
# dos tiles) de cada lado para ela. Por isso os limites abaixo sao a area
# das entradas (5%, 36%, 490 x 44%) expandida em 16 px por lado: as entradas
# continuam no mesmo lugar e com o mesmo tamanho, e o contorno as envolve.
#   left = 5% - 16   top = 36% - 16   width = 490 + 32   height = 44% + 32
# Se algum nome for cortado, aumente "width" (mantendo os +32 do contorno);
# para encolher a selecao, diminua.
+ boot_menu {
  left = 5%-16
  top = 36%-16
  width = 522
  height = 44%+32
  menu_pixmap_style = "boot_menu_box_*.png"
  item_font = "Public Sans Bold 21"
  item_color = "#dddddd"
  selected_item_color = "#ffffff"
  icon_width = 75
  icon_height = 75
  item_icon_space = 20
  item_height = 70
  item_padding = 3
  item_spacing = 13
  selected_item_pixmap_style = "select_*.png"
  scrollbar = false
}
